function addToCart() { const prodId = parseInt(document.getElementById('posSelectProduct').value); const priceType = document.getElementById('posPriceType').value; const qty = parseInt(document.getElementById('posQty').value); const shipping = parseFloat(document.getElementById('posShippingCost').value) || 0; if (!prodId) return alert('اختر صنفاً أولاً!'); const item = inventory.find(p => p.id === prodId); if (qty > item.qty) return alert('الكمية المطلوبة أكبر من المتاح بالمخزن!'); const price = priceType === 'customer' ? item.priceCustomer : item.priceMechanic; const existingCartItem = cart.find(c => c.id === prodId && c.priceType === priceType); if (existingCartItem) { if (existingCartItem.qty + qty > item.qty) { return alert('الكمية الإجمالية في السلة تتجاوز المتاح بالمخزن!'); } existingCartItem.qty += qty; existingCartItem.shipping += (item.category === 'عجل' ? shipping : 0); existingCartItem.total = (existingCartItem.qty * price) + existingCartItem.shipping; } else { const itemShipping = item.category === 'عجل' ? shipping : 0; cart.push({ id: item.id, name: item.name, category: item.category, cost: item.cost, price: price, priceType: priceType, qty: qty, shipping: itemShipping, total: (qty * price) + itemShipping }); } populatePosDropdown(); document.getElementById('posQty').value = 1; document.getElementById('posShippingCost').value = 0; renderCart(); } function renderCart() { const tbody = document.getElementById('cartTable'); tbody.innerHTML = ''; let grandTotal = 0; cart.forEach((item, index) => { grandTotal += item.total; tbody.innerHTML += ` ${item.name} ${item.category} ${item.price.toFixed(2)} ج.م ${item.qty} ${item.shipping.toFixed(2)} ج.م ${item.total.toFixed(2)} ج.م `; }); document.getElementById('cartTotal').textContent = `${grandTotal.toFixed(2)} ج.م`; calcRemaining(); } function removeFromCart(index) { cart.splice(index, 1); renderCart(); } function togglePayInputs() { const type = document.getElementById('payType').value; const container = document.getElementById('partialPayContainer'); if (type === 'partial') { container.classList.remove('hidden'); } else { container.classList.add('hidden'); } calcRemaining(); } function calcRemaining() { const total = cart.reduce((sum, item) => sum + item.total, 0); const payType = document.getElementById('payType').value; let paid = 0; if (payType === 'cash') { paid = total; } else if (payType === 'credit') { paid = 0; } else { paid = parseFloat(document.getElementById('paidAmount').value) || 0; } const remaining = total - paid; document.getElementById('remainingAmount').textContent = `${(remaining > 0 ? remaining : 0).toFixed(2)} ج.م`; } function checkoutAndPrint() { if (cart.length === 0) return alert('السلة فارغة!'); const custName = document.getElementById('custName').value.trim(); const custPhone = document.getElementById('custPhone').value.trim(); const payType = document.getElementById('payType').value; if ((payType === 'credit' || payType === 'partial') && !custName) { return alert('يرجى كتابة اسم الزبون عند الشراء بالآجل!'); } const total = cart.reduce((sum, item) => sum + item.total, 0); let paid = 0; if (payType === 'cash') paid = total; else if (payType === 'credit') paid = 0; else paid = parseFloat(document.getElementById('paidAmount').value) || 0; const remaining = total - paid; // خصم الكميات من المخزن cart.forEach(c => { const invItem = inventory.find(i => i.id === c.id); if (invItem) invItem.qty -= c.qty; }); localStorage.setItem('bz_inventory_final', JSON.stringify(inventory)); // تسجيل المبيعات const saleRecord = { invoiceId: invoiceCounter, date: new Date().toLocaleString('ar-EG'), customerName: custName || 'عميل نقدي', customerPhone: custPhone || '-', items: [...cart], total: total, paid: paid, remaining: remaining, payType: payType }; sales.push(saleRecord); localStorage.setItem('bz_sales_final', JSON.stringify(sales)); // تسجيل الآجل إن وجد if (remaining > 0) { const debtItem = debts.find(d => d.name === custName); if (debtItem) { debtItem.totalDebt += remaining; debtItem.remaining += remaining; if (custPhone) debtItem.phone = custPhone; } else { debts.push({ id: Date.now(), name: custName, phone: custPhone || '-', totalDebt: remaining, paid: 0, remaining: remaining }); } localStorage.setItem('bz_debts_final', JSON.stringify(debts)); } // تجهيز الطباعة document.getElementById('invId').textContent = `#${invoiceCounter}`; document.getElementById('invCustName').textContent = custName || 'عميل نقدي'; document.getElementById('invCustPhone').textContent = custPhone || '-'; document.getElementById('invDate').textContent = saleRecord.date; document.getElementById('invPayType').textContent = payType === 'cash' ? 'كاش' : (payType === 'credit' ? 'آجل' : 'جزئي'); document.getElementById('invTotal').textContent = `${total.toFixed(2)} ج.م`; document.getElementById('invPaid').textContent = `${paid.toFixed(2)} ج.م`; document.getElementById('invRemaining').textContent = `${(remaining > 0 ? remaining : 0).toFixed(2)} ج.م`; const invItemsTbody = document.getElementById('invItems'); invItemsTbody.innerHTML = ''; let hasBikes = false; cart.forEach(item => { if (item.category === 'عجل') hasBikes = true; invItemsTbody.innerHTML += ` ${item.name} ${item.price.toFixed(2)} ${item.qty} ${item.shipping.toFixed(2)} ${item.total.toFixed(2)} `; }); // إظهار مربع الضمان فقط في حالة وجود دراجة const warrantyBox = document.getElementById('invWarrantyBox'); if (hasBikes) { warrantyBox.classList.remove('hidden'); } else { warrantyBox.classList.add('hidden'); } // تحديث عداد الفواتير وإعادة الضبط invoiceCounter++; localStorage.setItem('bz_invoice_counter', invoiceCounter); window.print(); cart = []; document.getElementById('custName').value = ''; document.getElementById('custPhone').value = ''; document.getElementById('paidAmount').value = ''; document.getElementById('payType').value = 'cash'; togglePayInputs(); renderCart(); renderInventory(); } function renderDebts() { const tbody = document.getElementById('debtsTable'); tbody.innerHTML = ''; debts.forEach((d) => { tbody.innerHTML += ` ${d.name} ${d.phone} ${d.totalDebt.toFixed(2)} ج.م ${d.paid.toFixed(2)} ج.م ${d.remaining.toFixed(2)} ج.م `; }); } function payDebt(debtId) { const debt = debts.find(d => d.id === debtId); if (!debt) return; const amountStr = prompt(`المبلغ المتبقي على العميل (${debt.name}) هو ${debt.remaining.toFixed(2)} ج.م.\nأدخل المبلغ المراد تحصيله:`); const amount = parseFloat(amountStr); if (amount && amount > 0) { if (amount > debt.remaining) { return alert('المبلغ المدفوع أكبر من باقي الدين!'); } debt.paid += amount; debt.remaining -= amount; localStorage.setItem('bz_debts_final', JSON.stringify(debts)); renderDebts(); alert('تم تسجيل التحصيل بنجاح!'); } } function calculateReports() { let totalSales = 0; let totalProfit = 0; sales.forEach(s => { totalSales += s.total; s.items.forEach(item => { const profitPerUnit = item.price - item.cost; totalProfit += (profitPerUnit * item.qty); }); }); const totalDebts = debts.reduce((sum, d) => sum + d.remaining, 0); const stockValue = inventory.reduce((sum, p) => sum + (p.cost * p.qty), 0); document.getElementById('repTotalSales').textContent = `${totalSales.toFixed(2)} ج.م`; document.getElementById('repTotalProfit').textContent = `${totalProfit.toFixed(2)} ج.م`; document.getElementById('repTotalDebts').textContent = `${totalDebts.toFixed(2)} ج.م`; document.getElementById('repStockValue').textContent = `${stockValue.toFixed(2)} ج.م`; }
